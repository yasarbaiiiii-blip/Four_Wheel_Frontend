Step 1: Update GeoJSON Feature Generation in backendTrajectoryPreview.ts

Split the path into finished and unfinished features based on the current waypoint or progress:

// In src/utils/backendTrajectoryPreview.ts
export function buildSplitBackendTrajectoryCollection(
  points: TrajectoryPreviewPoint[],
  currentProgressIndex: number = 0
): LineFeatureCollection {
  const allCoords = points.map(p => [p.longitude, p.latitude]);
  const splitIdx = Math.max(0, Math.min(allCoords.length, currentProgressIndex));

  const finishedCoords = allCoords.slice(0, splitIdx + 1);
  const unfinishedCoords = allCoords.slice(splitIdx);

  const features = [];

  // Finished Trajectory LineString
  if (finishedCoords.length >= 2) {
    features.push({
      type: "Feature",
      properties: { kind: "line", lineStatus: "finished" },
      geometry: { type: "LineString", coordinates: finishedCoords }
    });
  }

  // Unfinished Trajectory LineString
  if (unfinishedCoords.length >= 2) {
    features.push({
      type: "Feature",
      properties: { kind: "line", lineStatus: "unfinished" },
      geometry: { type: "LineString", coordinates: unfinishedCoords }
    });
  }

  return { type: "FeatureCollection", features };
}
